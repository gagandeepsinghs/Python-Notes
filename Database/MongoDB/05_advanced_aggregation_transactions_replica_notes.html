$unwind Array Expansion
db.orders.aggregate([ { $unwind: "$items" }, { $project: { customer: 1, product: "$items.product", qty: "$items.qty" } } ])

$lookup Foreign Collection Join
db.orders.aggregate([ { $lookup: { from: "customers", localField: "customerId", foreignField: "_id", as: "customer" } } ])

$facet Multi-Pipeline Analytics
db.products.aggregate([ { $facet: { byCategory: [ { $group: { _id: "$category", count: { $sum: 1 } } } ], topProducts: [ { $sort: { sales: -1 } }, { $limit: 5 } ] } } ])

Aggregation Expressions ($multiply, $year)
db.orders.aggregate([ { $project: { customer: 1, total: { $multiply: ["$price", "$quantity"] }, year: { $year: "$createdAt" } } } ])

Aggregation Output via $out & $merge
db.orders.aggregate([ { $group: { _id: "$customer", total: { $sum: "$amount" } } }, { $merge: { into: "customer_totals" } } ])

ACID Transactions — Session Pattern
const session = db.getMongo().startSession(); const sdb = session.getDatabase("college"); try { session.startTransaction(); sdb.students.updateOne( { name: "Aman" }, { $inc: { balance: -100 } } ); sdb.fees.updateOne( { student: "Aman" }, { $inc: { paid: 100 } } ); session.commitTransaction(); } catch (e) { session.abortTransaction(); } finally { session.endSession(); }
