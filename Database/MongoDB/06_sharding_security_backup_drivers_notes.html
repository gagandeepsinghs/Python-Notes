Python PyMongo — Full CRUD
from pymongo import MongoClient client = MongoClient("mongodb://localhost:27017") db = client["college"] students = db["students"] // Insert students.insert_one({"name": "Aman", "age": 22, "course": "MCA"}) // Read student = students.find_one({"name": "Aman"}) print(student) // Update students.update_one({"name": "Aman"}, {"$set": {"city": "Ludhiana"}}) // Delete students.delete_one({"name": "Aman"}) client.close()

Python Aggregation Pipeline
pipeline = [ {"$match": {"course": "MCA"}}, {"$group": {"_id": "$city", "count": {"$sum": 1}}}, {"$sort": {"count": -1}} ] for row in students.aggregate(pipeline): print(row)

Node.js Native Driver Full CRUD
const { MongoClient } = require("mongodb"); const client = new MongoClient(process.env.MONGO_URI); async function main() { await client.connect(); const db = client.db("college"); const students = db.collection("students"); await students.insertOne({ name: "Aman", age: 22 }); console.log(await students.find({}).toArray()); await students.updateOne({ name: "Aman" }, { $set: { city: "Ludhiana" } }); await students.deleteOne({ name: "Aman" }); await client.close(); } main().catch(console.error);
